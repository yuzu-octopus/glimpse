```ts
// src/shared/config.ts
export const ColumnSchema = z.object({
  size: z.enum(['small','full']),
  span: z.number().int().min(1).max(12).optional(),
  widgets: z.array(WidgetSchema),
});
export function resolveSpan(columns: Column[]): number[] {
  if (columns.every(c => typeof c.span === 'number')) return columns.map(c => c.span!);
  const hasExplicit = columns.some(c => c.span);
  if (hasExplicit) throw new Error('mix of explicit span and size not allowed');
  const sizes = columns.map(c => c.size);
  if (sizes.length === 1) return sizes[0] === 'small' ? [3] : [12];
  if (sizes.length === 2) {
    if (sizes[0]==='full' && sizes[1]==='full') return [6,6];
    if (sizes.includes('small') && sizes.includes('full')) return sizes[0]==='full' ? [9,3] : [3,9];
  }
  if (sizes.length === 3) {
    if (sizes.every(s=>s==='full')) return [4,4,4];
    if (sizes.filter(s=>s==='full').length===1) {
      const idx = sizes.indexOf('full');
      const out = [3,3,3]; out[idx]=6; return out;
    }
  }
  return columns.map(() => 4);
}
```

```ts
export async function fetchWithRetry(ctx: WidgetFetchContext, url: string, httpOpts: HttpOptions, retryOpts={retries:3, baseDelay:500, factor:2}): Promise<Response> {
  let last: Response|Error;
  for(let attempt=0; attempt<=retryOpts.retries; attempt++){
    try{
      const res = await ctx.fetch(url, {headers:httpOpts.headers, signal: AbortSignal.timeout(httpOpts.timeoutMs??15000)} as any);
      if(res.ok) return res;
      if(![403,429,500,502,503,504].includes(res.status)) throw new Error(`HTTP ${res.status} for ${url}`);
      const retryAfter = res.headers.get('Retry-After');
      let delay = retryOpts.baseDelay * Math.pow(retryOpts.factor, attempt) + Math.random()*100;
      if(retryAfter){ const secs = Number(retryAfter); if(!isNaN(secs)) delay = secs*1000; else { const d=Date.parse(retryAfter); if(!isNaN(d)) delay = Math.max(0, d - Date.now()); } }
      last = res;
      if(attempt < retryOpts.retries) await new Promise(r=>setTimeout(r, delay));
      else throw new Error(`HTTP ${res.status} for ${url}`);
    } catch(e){
      last = e as Error;
      if(attempt < retryOpts.retries) await new Promise(r=>setTimeout(r, retryOpts.baseDelay*Math.pow(retryOpts.factor,attempt)));
      else throw e;
    }
  }
  throw last;
}
export async function fetchJson<T>(ctx,url,opts={}){ const res=await fetchWithRetry(ctx,url,opts); return res.json() as Promise<T>; }
```
